set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_target(bootstrap_registry ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build-bootstrap.py"
    --root "${CMAKE_CURRENT_SOURCE_DIR}" --output "${generated}"
    --git "${GIT_EXECUTABLE}" --compiler "${CMAKE_CXX_COMPILER}"
    --compiler-version "${CMAKE_CXX_COMPILER_VERSION}"
    --sdk "${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}" --configuration "$<CONFIG>"
    --compile-flags "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}"
    --link-flags "${CMAKE_EXE_LINKER_FLAGS} ${CMAKE_EXE_LINKER_FLAGS_RELEASE}"
    --cmake-version "${CMAKE_VERSION}" --generator "${CMAKE_GENERATOR}"
  BYPRODUCTS "${generated}/bootstrap_registry.h" "${generated}/build-identity.json"
  COMMENT "Bind current Git revision and native input bytes" VERBATIM)

function(bootstrap_executable target)
  add_executable(${target} source/apps/disked/main.cpp source/apps/disked/cli.cpp source/apps/disked/command_registry.cpp
    source/apps/disked/disked.manifest source/providers/fake/bootstrap.cpp source/providers/fake/graph.cpp source/providers/fake/health_observer.cpp
    source/platform/windows/entry.cpp source/platform/windows/invocation.cpp source/platform/windows/fake_worker.cpp source/platform/windows/output.cpp source/platform/windows/memory_budget.cpp source/platform/windows/console/terminal.cpp source/platform/windows/ui/gui.cpp)
  add_dependencies(${target} bootstrap_registry)
  target_include_directories(${target} PRIVATE "${generated}" source/providers/fake source/platform/windows source/platform/windows/console source/platform/windows/ui source/apps/disked)
  target_link_libraries(${target} PRIVATE disked_tui disked_gui disked_shell disked_image_actions disked_file_acquisition disked_acquisition_observation disked_health_observation)
  target_sources(${target} PRIVATE source/platform/windows/acquisition_worker.cpp)
  if(target STREQUAL "disked_image_test")
    target_sources(${target} PRIVATE source/providers/image/file_capture.cpp)
    target_include_directories(${target} PRIVATE source/providers/image)
    target_link_libraries(${target} PRIVATE disked_map_observation disked_local_file)
    target_compile_definitions(${target} PRIVATE DISKED_IMAGE_COMMAND_TESTING=1 DISKED_IMAGE_CAPTURE_TESTING=1)
  else()
    target_link_libraries(${target} PRIVATE disked_image_capture)
  endif()
  set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /GS /sdl /guard:cf)
  target_link_options(${target} PRIVATE /SUBSYSTEM:CONSOLE,10.00 /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
endfunction()
bootstrap_executable(disked)
