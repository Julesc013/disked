   OR NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL expected_sdk
   OR NOT actual_toolset STREQUAL expected_toolset)
  message(FATAL_ERROR "Use the pinned windows-bootstrap preset/toolchain. Observed: Windows=${WIN32}; MSVC=${MSVC}; pointer=${CMAKE_SIZEOF_VOID_P}; platform=${CMAKE_VS_PLATFORM_NAME}; compiler=${CMAKE_CXX_COMPILER_VERSION}; SDK=${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}; toolset=${actual_toolset}")

function(bootstrap_executable target)
  add_executable(${target} source/apps/disked/main.cpp source/apps/disked/app.cpp source/apps/disked/cli/cli.cpp source/apps/disked/command_registry.cpp
    source/apps/disked/disked.manifest source/providers/fake/bootstrap.cpp source/providers/fake/graph.cpp source/providers/fake/health_observer.cpp
    source/platform/windows/entry.cpp source/platform/windows/invocation.cpp source/platform/windows/fake_worker.cpp source/platform/windows/output.cpp source/platform/windows/memory_budget.cpp source/platform/windows/console/terminal.cpp source/platform/windows/ui/gui.cpp)
  add_dependencies(${target} bootstrap_registry)
  if(target STREQUAL "disked_recursive_disabled" OR target STREQUAL "disked_tree_test")
    target_compile_definitions(${target} PRIVATE DISKED_RECURSIVE_DISABLED=1)
  else()
    if(target STREQUAL "disked_recursive_fault")
      target_link_libraries(${target} PRIVATE disked_fat_tree_actions_fault disked_fat_tree_reconnect disked_fat_tree_controller_fault disked_fat_tree_extraction_worker_fault)
      target_compile_definitions(${target} PRIVATE DISKED_WORKER_TEST_STORE_FAULTS=1)
    else()
      target_link_libraries(${target} PRIVATE disked_fat_tree_actions disked_fat_tree_reconnect disked_fat_tree_controller disked_fat_tree_extraction_worker)
    endif()
  endif()
  if(target STREQUAL "disked_tree_test")
    target_sources(${target} PRIVATE source/platform/windows/storage/fat_tree_sessions.cpp)
    target_link_libraries(${target} PRIVATE disked_fat_tree_session disked_fat_tree_worker_fault)
    target_compile_definitions(${target} PRIVATE DISKED_FAT_TREE_COMMAND_TESTING=1)
  else()
    target_link_libraries(${target} PRIVATE disked_fat_tree_sessions disked_fat_tree_worker)
  endif()
  target_include_directories(${target} PRIVATE "${generated}")
  target_link_libraries(${target} PRIVATE disked_invocation disked_tui disked_gui disked_shell disked_image_actions disked_file_acquisition disked_acquisition_observation disked_health_observation disked_file_acquisition_export disked_file_verification_case_export disked_report_observation disked_file_image_verification disked_verification_observation)

    target_sources(${target} PRIVATE source/platform/windows/storage/fat_reader.cpp)
    target_link_libraries(${target} PRIVATE disked_fat_worker_fault)

  else()
    target_link_libraries(${target} PRIVATE disked_fat_reader)

  get_target_property(kind ${target} TYPE)

  set(headers "")
  foreach(source IN LISTS target_sources)
    if(source MATCHES "^source/.*\\.(c|cpp)$")
      string(REGEX REPLACE "\\.(c|cpp)$" ".h" header "${source}")
      if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${header}")
        list(APPEND headers "${header}")
      endif()
    endif()
  endforeach()
  if(target STREQUAL "disked_map_observation")
    list(APPEND headers source/runtime/capture/capture_port.h)
  elseif(target STREQUAL "disked_presentation")
    list(APPEND headers source/runtime/presentation/text_input.h)
  elseif(target STREQUAL "disked_fat")
    list(APPEND headers source/portable/fat/fat.h)
  elseif(target STREQUAL "disked_exfat")
    list(APPEND headers source/portable/exfat/exfat.h)
  endif()
  if(headers)
    list(REMOVE_DUPLICATES headers)
    if(kind STREQUAL "STATIC_LIBRARY")
      set(header_scope PUBLIC)
    else()
      set(header_scope PRIVATE)
    endif()
    target_sources(${target} ${header_scope} FILE_SET disked_internal TYPE HEADERS
      BASE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/source" FILES ${headers})
  endif()
endforeach()

# Project current literal compile-usage edges from CMake itself. The checker
# must not mistake transitive/order build dependencies for header visibility.
set(header_usage "{\"schema\":\"org.disked.private-header-usage/1\",\"targets\":{}}")
foreach(target IN LISTS disked_native_targets)
  set(target_usage "{\"direct\":[],\"interface\":[],\"headers\":[],\"interface_headers\":[]}")
  foreach(pair "direct|LINK_LIBRARIES" "interface|INTERFACE_LINK_LIBRARIES")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 key)
    list(GET fields 1 property)
    get_target_property(libraries ${target} ${property})
    set(index 0)
    foreach(library IN LISTS libraries)
      if(key STREQUAL "interface" AND library MATCHES "^\\$<LINK_ONLY:.*>$")
        continue()
      endif()
      if(library MATCHES "\\$<")
        message(FATAL_ERROR "Private header-usage projection needs an explicit rule for ${target}: ${library}")
      endif()
      if(TARGET ${library})
        string(JSON target_usage SET "${target_usage}" ${key} ${index} "\"${library}\"")
        math(EXPR index "${index}+1")
      endif()
    endforeach()
  endforeach()
  foreach(pair "headers|HEADER_SETS" "interface_headers|INTERFACE_HEADER_SETS")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 key)
    list(GET fields 1 property)
    get_target_property(sets ${target} ${property})
    set(index 0)
    if("disked_internal" IN_LIST sets)
      get_target_property(headers ${target} HEADER_SET_disked_internal)
      foreach(header IN LISTS headers)
        file(RELATIVE_PATH relative_header "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
        if(NOT relative_header MATCHES "^source/.*\\.h$")
          message(FATAL_ERROR "Internal header outside source ownership: ${header}")
        endif()
        string(JSON target_usage SET "${target_usage}" ${key} ${index} "\"${relative_header}\"")
        math(EXPR index "${index}+1")
      endforeach()
    endif()
  endforeach()
  string(JSON header_usage SET "${header_usage}" targets ${target} "${target_usage}")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/private-header-usage.json" CONTENT "${header_usage}\n")
