  message(FATAL_ERROR "Only the selected Windows x64 MSVC source candidate is in scope")
endif()

get_filename_component(DISKED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT IS_DIRECTORY "${USK_SOURCE_DIR}")
  message(FATAL_ERROR "Explicit verified USK_SOURCE_DIR required")
endif()
file(READ "${DISKED_ROOT}/external/universal-setup/native-source-lock.json" LOCK)
string(JSON ROWS LENGTH "${LOCK}" files)
math(EXPR LAST "${ROWS}-1")
foreach(INDEX RANGE ${LAST})
  string(JSON PATH GET "${LOCK}" files ${INDEX} path)
  string(JSON EXPECTED GET "${LOCK}" files ${INDEX} sha256)
  string(JSON EXPECTED_SIZE GET "${LOCK}" files ${INDEX} bytes)
  file(SHA256 "${USK_SOURCE_DIR}/${PATH}" HASH)
  file(SIZE "${USK_SOURCE_DIR}/${PATH}" SIZE)
  if(NOT "sha256:${HASH}" STREQUAL "${EXPECTED}" OR NOT SIZE EQUAL EXPECTED_SIZE)
    message(FATAL_ERROR "Pinned source changed: ${PATH}")
  endif()
endforeach()
function(lock_paths KEY RESULT)
  string(JSON COUNT LENGTH "${LOCK}" ${KEY})
  math(EXPR LAST "${COUNT}-1")
  set(VALUES)
  foreach(INDEX RANGE ${LAST})
    string(JSON PATH GET "${LOCK}" ${KEY} ${INDEX})
    list(APPEND VALUES "${USK_SOURCE_DIR}/${PATH}")
  endforeach()
  set(${RESULT} "${VALUES}" PARENT_SCOPE)
endfunction()
lock_paths(core_sources CORE)
lock_paths(zlib_sources INFLATE)
lock_paths(private_include_dirs PRIVATE_INCLUDES)
add_library(disked_setup_inflate OBJECT ${INFLATE})
target_include_directories(disked_setup_inflate PRIVATE "${USK_SOURCE_DIR}/external/zlib/upstream")
target_compile_definitions(disked_setup_inflate PRIVATE Z_PREFIX)
add_library(disked_setup_source_candidate STATIC ${CORE} $<TARGET_OBJECTS:disked_setup_inflate>)
target_include_directories(disked_setup_source_candidate PUBLIC "${USK_SOURCE_DIR}/include" PRIVATE ${PRIVATE_INCLUDES})
target_compile_definitions(disked_setup_source_candidate PRIVATE Z_PREFIX)
target_compile_options(disked_setup_source_candidate PRIVATE /W4 /WX)
add_executable(setup_source_probe "${CMAKE_CURRENT_LIST_DIR}/source_probe.c")
target_compile_options(setup_source_probe PRIVATE /W4 /WX)
target_link_libraries(setup_source_probe PRIVATE disked_setup_source_candidate)
option(DISKED_BUILD_HOST_FIXTURE "Build product-bound read-only H fixture" OFF)
if(DISKED_BUILD_HOST_FIXTURE)
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${DISKED_ROOT}"
    OUTPUT_VARIABLE HOST_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  string(LENGTH "${HOST_REVISION}" REVISION_BYTES)
  if(NOT HOST_REVISION MATCHES "^[0-9a-f]+$" OR NOT REVISION_BYTES EQUAL 40)
    message(FATAL_ERROR "Exact host source revision required")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain WORKING_DIRECTORY "${DISKED_ROOT}"
    OUTPUT_VARIABLE HOST_CHANGES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  set(HOST_SOURCE_STATE clean)
  if(HOST_CHANGES)
    set(HOST_SOURCE_STATE modified)
  endif()
  string(JSON HOST_SETUP_REVISION GET "${LOCK}" revision)
  add_executable(disked_setup_host_fixture "${CMAKE_CURRENT_LIST_DIR}/source_probe.c")
  target_compile_options(disked_setup_host_fixture PRIVATE /W4 /WX)
  target_compile_definitions(disked_setup_host_fixture PRIVATE DISKED_SETUP_HOST_FIXTURE
    DISKED_HOST_REVISION="${HOST_REVISION}" DISKED_HOST_SOURCE_STATE="${HOST_SOURCE_STATE}"
    DISKED_HOST_SETUP_REVISION="${HOST_SETUP_REVISION}")
  target_link_libraries(disked_setup_host_fixture PRIVATE disked_setup_source_candidate)
endif()
