cmake_minimum_required(VERSION 3.27)
project(DirectoryProbe LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(worker_directory_probe "${ROOT}/tests/operation/directory_probe.cpp" "${ROOT}/source/runtime/command/json.cpp")
target_include_directories(worker_directory_probe PRIVATE "${ROOT}/source/platform/windows" "${ROOT}/source/runtime/command" "${ROOT}/source/portable/hash")
target_compile_definitions(worker_directory_probe PRIVATE _WIN32_WINNT=0x0A00)
set_target_properties(worker_directory_probe PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(worker_directory_probe PRIVATE /W4 /WX /permissive- /utf-8 /GS /sdl /guard:cf)
target_link_options(worker_directory_probe PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
